START
  │
  ├─→ [INPUT PARAMETERS FROM FRONTEND]
  │   │
  │   └─→ Input: JSON/Dictionary object containing:
  │       │
  │       ├─→ store_code: String (e.g., "RHN", "HBT", "HDG", etc.)
  │       │
  │       ├─→ store_target: Number (store's revenue target)
  │       │
  │       ├─→ store_fp_ratio_target: Number (target full-price ratio, e.g., 0.7 for 70%)
  │       │
  │       ├─→ query_date: Object {
  │       │   ├─→ from_date: Date (start date for query period)
  │       │   └─→ to_date: Date (end date for query period)
  │       │   }
  │       │
  │       └─→ employees: Dictionary/Array of employee objects [
  │           {
  │             ├─→ employee_code: String (unique employee identifier)
  │             ├─→ full_name: String (employee's full name)
  │             ├─→ seniority: Number (tenure in years)
  │             ├─→ personal_target: Number (employee's personal sales target)
  │             ├─→ working_day_count: Number (days worked in the period)
  │             ├─→ is_manager: Boolean (true if employee is store manager)
  │             └─→ is_probation: Boolean (true if employee is on probation)
  │           },
  │           ...
  │         ]
  │
  ├─→ [DERIVED/CALCULATED PARAMETERS]
  │   │
  │   ├─→ month = EXTRACT_MONTH(query_date.from_date)
  │   ├─→ year = EXTRACT_YEAR(query_date.from_date)
  │   ├─→ all_sales_df (queried from database for the period - SINGLE QUERY)
  │   │   // Query returns DataFrame with columns:
  │   │   // ['sale_id', 'employee_code', 'bill_number', 'store_id', 'sale_date', 'sale_time',
  │   │   //  'revenue_with_vat', 'revenue_before_vat', 'discount_rate',
  │   │   //  'is_jewelry', 'vendor_code', 'category', 'department']
  │   │   // Includes: ALL items (jewelry and non-jewelry), all stores, all vendors
  │   │   // Filtered: Only paid transactions
  │   │   // Ordered by: employee_code, sale_date, sale_time, bill_number (chronological order)
  │   │   // NOTE: employee_code is the HR code, not employee_id (sid in retailpro)
  │   │   // NOTE: bill_number groups items from the same transaction/receipt
  │   └─→ employee sales data (queried per employee from all_sales_df)
  │
  ├─→ [VALIDATION]
  │   │
  │   └─→ IF month IS NULL OR year IS NULL OR employees IS NULL OR len(employees) == 0:
  │       └─→ RETURN {
  │           success: false,
  │           error: "Missing required fields: month, year, or employees list"
  │       }
  │
  └─→ [PERSONAL COMMISSION CALCULATION PROCESS]

FUNCTION calculate_all_commissions(request_data):

    // 1. EXTRACT DATA FROM FRONTEND REQUEST
    month = EXTRACT_MONTH(request_data.query_date.from_date)
    year = EXTRACT_YEAR(request_data.query_date.from_date)
    store_code = request_data.store_code  // Store identifier
    employees = request_data.employees  // List of employee dictionaries

    // 2. VALIDATE INPUT
    IF month IS NULL OR year IS NULL OR employees IS NULL OR len(employees) == 0:
        RETURN {
            success: false,
            error: "Missing required fields: month, year, or employees list"
        }

    // 3. GET ALL SALES DATA AS PANDAS DATAFRAME (SINGLE QUERY)
    // Query ALL sales (jewelry + non-jewelry) - returns DataFrame with columns:
    // ['sale_id', 'employee_code', 'bill_number', 'store_id', 'sale_date', 'sale_time',
    //  'revenue_with_vat', 'revenue_before_vat', 'discount_rate',
    //  'is_jewelry', 'vendor_code', 'category', 'department']
    // Includes: ALL items (jewelry and non-jewelry), all stores, all vendors
    // Filtered: Only paid transactions
    // Ordered by: employee_code, sale_date, sale_time, bill_number (chronological order)
    // NOTE: employee_code is the HR code, not employee_id (sid in retailpro)
    // NOTE: bill_number groups items from the same transaction/receipt
    all_sales_df = query_all_sales(month, year)
    
    // 4. PREPARE RESULTS LIST
    results = []
    
    // 5. PROCESS EACH EMPLOYEE
    FOR EACH employee_data IN employees:
        employee_code = employee_data['employee_code']
        target = employee_data['personal_target']
        employee_name = employee_data.get('full_name', '')
        department = employee_data.get('department', '')

        // Validate employee data
        IF employee_code IS NULL OR target IS NULL:
            results.append({
                'employee_code': employee_code,
                'fullname': employee_name,
                'store_code': store_code,
                'commission_fp': 0,
                'commission_discount': 0,
                'commission_jewelry': 0,
                'commission_vhernier': 0,
                'commission_rosa_maria': 0,
                'commission_100_and_below_fp': 0,
                'commission_100_and_below_discount': 0,
                'commission_100_and_below': 0,
                'commission_over_100': 0,
                'total': 0
            })
            CONTINUE

        // Filter ALL sales for this specific employee (by HR code)
        // Includes ALL items: jewelry + non-jewelry, all stores, all vendors
        employee_sales_df = all_sales_df[all_sales_df['employee_code'] == employee_code].copy()

        // Check if employee has any sales at all
        IF len(employee_sales_df) == 0:
            results.append({
                'employee_code': employee_code,
                'fullname': employee_name,
                'store_code': store_code,
                'commission_fp': 0,
                'commission_discount': 0,
                'commission_jewelry': 0,
                'commission_vhernier': 0,
                'commission_rosa_maria': 0,
                'commission_100_and_below_fp': 0,
                'commission_100_and_below_discount': 0,
                'commission_100_and_below': 0,
                'commission_over_100': 0,
                'total': 0
            })
            CONTINUE

        // Calculate TOTAL revenue WITH VAT from ALL sources for eligibility check
        // (includes jewelry + non-jewelry, all stores, all vendors)
        total_revenue_with_vat = employee_sales_df['revenue_with_vat'].sum()

        // Calculate achievement rate based on TOTAL revenue WITH VAT from ALL sources
        achievement_rate = (total_revenue_with_vat / target) * 100

        // SEPARATE SALES BY TYPE for commission calculation
        // This must be done BEFORE eligibility check since jewelry commission is paid regardless
        // Non-jewelry sales
        non_jewelry_df = employee_sales_df[employee_sales_df['is_jewelry'] == false].copy()

        // Jewelry sales
        jewelry_df = employee_sales_df[employee_sales_df['is_jewelry'] == true].copy()

        // Calculate non-jewelry revenue breakdown
        // IMPORTANT: Use revenue_before_vat (non-VAT) for commission calculations
        non_jewelry_fp_df = non_jewelry_df[non_jewelry_df['discount_rate'] <= 0.30]
        non_jewelry_fp_revenue = non_jewelry_fp_df['revenue_before_vat'].sum()  // Non-VAT for commission

        non_jewelry_disc_df = non_jewelry_df[non_jewelry_df['discount_rate'] > 0.30]
        non_jewelry_disc_revenue = non_jewelry_disc_df['revenue_before_vat'].sum()  // Non-VAT for commission

        // GROUP BY BILL to calculate bill-level totals (for >100% tier calculation)
        // A bill can contain multiple items, and we need to check at BILL level when target is reached
        bill_totals_df = employee_sales_df.groupby('bill_number', sort=False).agg({
            'revenue_with_vat': 'sum',
            'sale_date': 'first',
            'sale_time': 'first'
        }).reset_index()

        // Sort bills chronologically
        bill_totals_df = bill_totals_df.sort_values(['sale_date', 'sale_time'])

        // Calculate running total by BILL (WITH VAT) to find when 100% target was reached
        bill_totals_df['running_total'] = bill_totals_df['revenue_with_vat'].cumsum()

        // Find full-price NON-JEWELRY sales after reaching 100% target (for >100% tier bonus)
        fp_non_jewelry_after_target = 0

        IF achievement_rate > 100:
            // Find the BILL where target was first reached or exceeded
            target_reached_bills = bill_totals_df[bill_totals_df['running_total'] >= target]

            IF len(target_reached_bills) > 0:
                // Get the first bill that reached/exceeded target
                target_bill_number = target_reached_bills.iloc[0]['bill_number']
                target_bill_running_total = target_reached_bills.iloc[0]['running_total']
                target_bill_revenue = target_reached_bills.iloc[0]['revenue_with_vat']

                previous_total = target_bill_running_total - target_bill_revenue

                // Get ALL items from the target-reaching bill
                target_bill_items = employee_sales_df[employee_sales_df['bill_number'] == target_bill_number]

                // Filter items from target bill: FP, non-jewelry, NOT TIT/TVL
                target_bill_qualifying = target_bill_items[
                    (target_bill_items['discount_rate'] <= 0.30) &
                    (target_bill_items['is_jewelry'] == false) &
                    ~(target_bill_items['vendor_code'].isin(['TIT', 'TVL']))
                ]

                // If this bill pushed over target, apply 2% to ALL qualifying items in this bill
                IF previous_total < target:
                    fp_non_jewelry_after_target = target_bill_qualifying['revenue_before_vat'].sum()

                // Get all bills AFTER the target-reaching bill
                target_bill_index = bill_totals_df[bill_totals_df['bill_number'] == target_bill_number].index[0]
                bills_after_target = bill_totals_df.iloc[target_bill_index + 1:]

                // For each bill after target, find qualifying items
                IF len(bills_after_target) > 0:
                    bills_after_target_numbers = bills_after_target['bill_number'].tolist()

                    // Get all items from bills that came after target
                    items_after_target = employee_sales_df[
                        employee_sales_df['bill_number'].isin(bills_after_target_numbers)
                    ]

                    // Filter for: full-price, non-jewelry, NOT TIT/TVL
                    fp_non_jewelry_after_df = items_after_target[
                        (items_after_target['discount_rate'] <= 0.30) &
                        (items_after_target['is_jewelry'] == false) &
                        ~(items_after_target['vendor_code'].isin(['TIT', 'TVL']))
                    ]

                    // Add to fp_non_jewelry_after_target (use revenue_before_vat)
                    fp_non_jewelry_after_target = fp_non_jewelry_after_target + fp_non_jewelry_after_df['revenue_before_vat'].sum()

        // FINE JEWELRY COMMISSION (Completely independent of personal commission)
        // IMPORTANT: All jewelry commission calculations use revenue_before_vat (non-VAT revenue)
        // CRITICAL: Jewelry commission is paid REGARDLESS of achievement rate
        //           Even if employee doesn't reach 50% threshold, they receive jewelry commission
        //           Jewelry commission rates are FIXED and do NOT vary by achievement tier
        jewelry_commission = 0
        commission_rosa_maria = 0  // Track ROM EARRINGS commission separately
        commission_vhernier = 0  // Track VHN commission separately
        jewelry_breakdown = []

        IF len(jewelry_df) > 0:
            // Calculate jewelry commission by vendor and category
            // Rates: TVL/TIT = 500k per item, ROM EARRINGS = 3%, VHN = 1%, Others = 2%
            // NOTE: These commissions are paid regardless of personal achievement rate

            // 1. TVL and TIT vendors: Flat 500,000 VND per item (handle first to exclude from % calcs)
            tvl_tit_df = jewelry_df[
                (jewelry_df['vendor_code'] == 'TVL') |
                (jewelry_df['vendor_code'] == 'TIT')
            ]
            IF len(tvl_tit_df) > 0:
                tvl_tit_item_count = len(tvl_tit_df)
                tvl_tit_commission = tvl_tit_item_count * 500000
                jewelry_commission = jewelry_commission + tvl_tit_commission

                jewelry_breakdown.append({
                    'vendor': 'TVL/TIT',
                    'category': 'All items',
                    'rate': 'Flat 500,000 VND per item',
                    'item_count': tvl_tit_item_count,
                    'commission': tvl_tit_commission
                })

            // Filter out TVL/TIT for percentage-based calculations
            jewelry_percent_df = jewelry_df[
                ~jewelry_df['vendor_code'].isin(['TVL', 'TIT'])
            ]

            IF len(jewelry_percent_df) > 0:
                // 2. ROM EARRINGS: 3% on revenue (highest priority)
                rom_earrings_df = jewelry_percent_df[
                    (jewelry_percent_df['vendor_code'] == 'ROM') &
                    (jewelry_percent_df['category'] == 'EARRINGS')
                ]
                IF len(rom_earrings_df) > 0:
                    rom_earrings_revenue = rom_earrings_df['revenue_before_vat'].sum()  // Non-VAT revenue
                    rom_earrings_commission = rom_earrings_revenue * 0.03
                    commission_rosa_maria = rom_earrings_commission  // Track separately
                    jewelry_commission = jewelry_commission + rom_earrings_commission

                    jewelry_breakdown.append({
                        'vendor': 'ROM',
                        'category': 'EARRINGS',
                        'rate': '3%',
                        'revenue': rom_earrings_revenue,
                        'commission': rom_earrings_commission
                    })

                // 3. VHN vendor: 1% on revenue (excluding ROM EARRINGS already counted)
                vhn_df = jewelry_percent_df[
                    (jewelry_percent_df['vendor_code'] == 'VHN') &
                    ~((jewelry_percent_df['vendor_code'] == 'ROM') &
                      (jewelry_percent_df['category'] == 'EARRINGS'))
                ]
                IF len(vhn_df) > 0:
                    vhn_revenue = vhn_df['revenue_before_vat'].sum()  // Non-VAT revenue
                    vhn_commission = vhn_revenue * 0.01
                    commission_vhernier = vhn_commission  // Track separately
                    jewelry_commission = jewelry_commission + vhn_commission

                    jewelry_breakdown.append({
                        'vendor': 'VHN',
                        'category': 'All categories',
                        'rate': '1%',
                        'revenue': vhn_revenue,
                        'commission': vhn_commission
                    })

                // 4. Other vendors (not VHN, not ROM EARRINGS, not TVL/TIT): 2% on revenue
                other_vendors_df = jewelry_percent_df[
                    (jewelry_percent_df['vendor_code'] != 'VHN') &
                    ~((jewelry_percent_df['vendor_code'] == 'ROM') &
                      (jewelry_percent_df['category'] == 'EARRINGS'))
                ]
                IF len(other_vendors_df) > 0:
                    other_vendors_revenue = other_vendors_df['revenue_before_vat'].sum()  // Non-VAT revenue
                    other_vendors_commission = other_vendors_revenue * 0.02
                    jewelry_commission = jewelry_commission + other_vendors_commission

                    jewelry_breakdown.append({
                        'vendor': 'Other vendors (excluding VHN, TVL, TIT)',
                        'category': 'All categories',
                        'rate': '2%',
                        'revenue': other_vendors_revenue,
                        'commission': other_vendors_commission
                    })

        // PERSONAL COMMISSION (Non-Jewelry) - Calculate based on NON-CUMULATIVE tiers
        // IMPORTANT: All commission calculations use revenue_before_vat (non-VAT revenue)
        // NOTE: Personal commission requires achievement_rate >= 50%
        personal_commission = 0
        commission_fp = 0  // Track FP commission separately
        commission_discount = 0  // Track discount commission separately
        commission_over_100 = 0  // Track over 100% bonus separately
        commission_breakdown = []

        // TIER 1: 50% - 70% of target
        IF achievement_rate >= 50 AND achievement_rate < 70:
            // Calculate commission on non-VAT revenue
            commission_fp = non_jewelry_fp_revenue * 0.0025
            commission_discount = non_jewelry_disc_revenue * 0.00125
            personal_commission = commission_fp + commission_discount

            commission_breakdown.append({
                'tier': '50-70%',
                'full_price_rate': '0.25%',
                'discounted_rate': '0.125%',
                'full_price_sales': non_jewelry_fp_revenue,
                'discounted_sales': non_jewelry_disc_revenue,
                'commission': personal_commission
            })

        // TIER 2: 70% - 100% of target
        ELSE IF achievement_rate >= 70 AND achievement_rate < 100:
            // Calculate commission on non-VAT revenue
            commission_fp = non_jewelry_fp_revenue * 0.005
            commission_discount = non_jewelry_disc_revenue * 0.0025
            personal_commission = commission_fp + commission_discount

            commission_breakdown.append({
                'tier': '70-100%',
                'full_price_rate': '0.5%',
                'discounted_rate': '0.25%',
                'full_price_sales': non_jewelry_fp_revenue,
                'discounted_sales': non_jewelry_disc_revenue,
                'commission': personal_commission
            })

        // TIER 3: 100% of target and above
        ELSE IF achievement_rate >= 100:
            // Base commission at 100% tier rates (on ALL non-jewelry sales)
            // Calculate commission on non-VAT revenue
            commission_fp = non_jewelry_fp_revenue * 0.01
            commission_discount = non_jewelry_disc_revenue * 0.005
            commission_up_to_target = commission_fp + commission_discount
            personal_commission = commission_up_to_target

            commission_breakdown.append({
                'tier': '100%',
                'full_price_rate': '1%',
                'discounted_rate': '0.5%',
                'full_price_sales': non_jewelry_fp_revenue,
                'discounted_sales': non_jewelry_disc_revenue,
                'commission': commission_up_to_target
            })

            // TIER 4: Over 100% - Additional 2% bonus
            // Only on FP, non-jewelry, NOT TIT/TVL sales that occurred AFTER reaching 100% target
            // Calculate commission on non-VAT revenue
            IF achievement_rate > 100 AND fp_non_jewelry_after_target > 0:
                commission_over_100 = fp_non_jewelry_after_target * 0.02  // Non-VAT revenue

                commission_breakdown.append({
                    'tier': 'Over 100% (Excess Bonus)',
                    'full_price_rate': '2%',
                    'discounted_rate': '0%',
                    'total_excess_revenue': total_revenue_with_vat - target,
                    'fp_non_jewelry_after_target': fp_non_jewelry_after_target,
                    'commission': commission_over_100,
                    'note': 'Bonus only applies to full-price, non-jewelry (excluding TIT/TVL) transactions that occurred after reaching 100% target'
                })

        // Calculate commission components
        // commission_100_and_below broken into: FP, discount, and jewelry
        // commission_over_100: Tier 4 bonus only
        commission_100_and_below_fp = commission_fp
        commission_100_and_below_discount = commission_discount
        commission_100_and_below = personal_commission + jewelry_commission
        total_commission = commission_100_and_below + commission_over_100

        // Add result for this employee (for DataFrame row)
        results.append({
            'employee_code': employee_code,
            'fullname': employee_name,
            'store_code': store_code,
            'commission_fp': commission_fp,
            'commission_discount': commission_discount,
            'commission_jewelry': jewelry_commission,
            'commission_vhernier': commission_vhernier,
            'commission_rosa_maria': commission_rosa_maria,
            'commission_100_and_below_fp': commission_100_and_below_fp,
            'commission_100_and_below_discount': commission_100_and_below_discount,
            'commission_over_100': commission_over_100,
            'total': total_commission
        })
    
    // 6. CONVERT RESULTS TO DATAFRAME
    // Create pandas DataFrame with specified columns
    results_df = pd.DataFrame(results, columns=[
        'employee_code',
        'fullname',
        'store_code',
        'commission_fp',
        'commission_discount',
        'commission_jewelry',
        'commission_vhernier',
        'commission_rosa_maria',
        'commission_100_and_below_fp',
        'commission_100_and_below_discount',
        'commission_over_100',
        'total'
    ])

    // 7. RETURN DATAFRAME
    RETURN results_df

END FUNCTION

NOTES & CLARIFICATIONS:
═══════════════════════════════════════════════════════════════════════════

1. PERSONAL COMMISSION RATE STRUCTURE (Non-Jewelry Items):

   IMPORTANT: Commission tiers are NON-CUMULATIVE. The rate applies to ALL non-jewelry
   sales based on the achievement tier reached, NOT added to previous tiers.

   TIER 1: 50-70% Achievement
   • Full Price Sales (discount_rate ≤ 0.30): 0.25%
   • Discounted Sales (discount_rate > 0.30): 0.125%

   TIER 2: 70-100% Achievement
   • Full Price Sales: 0.5%
   • Discounted Sales: 0.25%

   TIER 3: 100% Achievement
   • Full Price Sales: 1%
   • Discounted Sales: 0.5%

   TIER 4: Over 100% Achievement (Additional Bonus)
   • Full Price Sales AFTER reaching target: 2%
   • Calculated at BILL level (not individual item level)
   • Steps:
     1. Group items by bill_number and sum revenue_with_vat per bill
     2. Use cumsum on bills (chronologically) to find which BILL reached 100% target
     3. For the target-reaching bill: Apply 2% to ALL qualifying items in that bill
     4. For all bills after target: Apply 2% to ALL qualifying items
   • Qualifying items must be: Full-price (discount_rate ≤ 0.30) AND Non-jewelry AND NOT TIT/TVL
   • Commission calculated on revenue_before_vat of qualifying items

2. FINE JEWELRY COMMISSION (Completely Independent):

   CRITICAL: Fine jewelry commission is COMPLETELY INDEPENDENT of personal
   commission and achievement rate. Jewelry commission is paid to ALL employees
   regardless of whether they reach the 50% threshold or not.

   IMPORTANT: Jewelry commission rates are FIXED and do NOT change based on
   achievement tiers. An employee with 20% achievement receives the same jewelry
   commission rates as an employee with 120% achievement.

   Commission rates are based on VENDOR CODE and CATEGORY:

   A. ROM EARRINGS (Highest Priority): 3%
      • Vendor Code: 'ROM'
      • Category: 'EARRINGS'
      • Rate: 3% on total revenue (both FP and discounted, non-VAT)

   B. VHN Vendor: 1%
      • Vendor Code: 'VHN'
      • Rate: 1% on total revenue (both FP and discounted, non-VAT)

   C. Other Vendors (Default): 2%
      • Any vendor code EXCEPT 'VHN'
      • Excludes ROM EARRINGS (already counted at 3%)
      • Rate: 2% on total revenue (both FP and discounted, non-VAT)

   D. TVL/TIT Vendors (Special Flat Rate): 500,000 VND per item
      • Vendor Code: 'TVL' or 'TIT'
      • Rate: Flat 500,000 VND per transaction/item
      • Applied PER ITEM, not based on revenue
      • NOTE: TIT/TVL jewelry items do NOT count toward the Over 100% bonus tier

   NOTES:
   • Jewelry commission is COMPLETELY INDEPENDENT of personal commission
   • NO achievement threshold required - paid to all employees with jewelry sales
   • Jewelry commission rates are FIXED - same at 0%, 55%, 80%, or 120% achievement
   • Jewelry commission is based on revenue_before_vat for percentage-based rates
   • All jewelry items count (both full-price and discounted)
   • TIT/TVL jewelry excluded from Over 100% bonus calculation (they get flat rate only)

3. ELIGIBILITY REQUIREMENTS:

   PERSONAL COMMISSION (Non-Jewelry):
   • Employee must reach at least 50% of their personal target to be eligible
   • If achievement_rate < 50%: No personal commission is awarded
   • If achievement_rate >= 50%: Employee receives personal commission based on their tier

   JEWELRY COMMISSION (Jewelry, TVL, TIT):
   • NO achievement threshold required
   • ALL employees receive jewelry commission if they have jewelry sales
   • Paid regardless of personal achievement rate (even if 0%)

4. TOTAL COMMISSION CALCULATION:

   total_commission = personal_commission + jewelry_commission

   Where:
   • personal_commission = Commission from non-jewelry sales based on tier
   • jewelry_commission = Commission from fine jewelry sales (if eligible)

5. ACHIEVEMENT RATE AND COMMISSION CALCULATION:

   ELIGIBILITY CALCULATION (uses revenue WITH VAT):
   achievement_rate = (total_revenue_with_vat / personal_target) × 100%

   Where:
   total_revenue_with_vat = SUM of all revenue_with_vat from ALL sales (jewelry + non-jewelry)

   IMPORTANT - TWO DIFFERENT REVENUE VALUES:

   A. FOR ELIGIBILITY/ACHIEVEMENT RATE (uses revenue WITH VAT):
      • Total revenue WITH VAT from ALL sources (jewelry + non-jewelry, all stores, all vendors)
      • Used to determine if employee reaches 50%, 70%, or 100% thresholds
      • Determines which commission tier the employee qualifies for

   B. FOR ACTUAL COMMISSION CALCULATION (uses revenue BEFORE VAT / non-VAT):
      • All commission amounts are calculated on revenue_before_vat (non-VAT revenue)
      • Personal commission: Applied to non_jewelry_fp_revenue and non_jewelry_disc_revenue (both non-VAT)
      • Jewelry commission: Applied to jewelry revenue_before_vat (non-VAT)
      • Over 100% bonus: Applied to fp_non_jewelry_after_target (non-VAT)

   EXAMPLE:
   • Employee has 100M revenue WITH VAT, 90M revenue BEFORE VAT
   • Target: 100M
   • Achievement rate: (100M / 100M) × 100% = 100% → Qualifies for 1% tier
   • Commission: 90M × 1% = 900,000 VND (calculated on non-VAT revenue)

6. INPUT STRUCTURE:

   Frontend sends a JSON/Dictionary object containing:
   • store_code: Store identifier (e.g., "RHN", "HBT")
   • store_target: Store's revenue target
   • store_fp_ratio_target: Target full-price ratio
   • query_date: {from_date, to_date} for the period
   • employees: Array of employee objects with:
     - employee_code: Unique identifier
     - full_name: Employee's full name
     - seniority: Tenure in years
     - personal_target: Individual sales target (for non-jewelry items)
     - working_day_count: Days worked in period
     - is_manager: Boolean flag for manager status
     - is_probation: Boolean flag for probation status

7. DATA ARCHITECTURE:

   SINGLE QUERY APPROACH:
   • All sales (jewelry + non-jewelry) are retrieved in ONE query
   • Ordered chronologically by employee_code, sale_date, sale_time, bill_number
   • Each row includes: bill_number, revenue_with_vat, revenue_before_vat, is_jewelry flag,
     vendor_code, category, discount_rate
   • bill_number groups multiple items from the same transaction/receipt

   WHY SINGLE QUERY:
   • Achievement rate based on total revenue WITH VAT requires all sales together
   • Over 100% bonus is order-sensitive (needs to know WHICH bills came after target)
   • Bill-level grouping enables proper target-reaching calculation
   • Easier to filter by type (FP/discount, jewelry/non-jewelry, TIT/TVL) for each tier
   • Maintains chronological integrity of sales data at both bill and item level

   BILL-LEVEL LOGIC FOR OVER 100% TIER:
   • A bill can contain multiple items (jewelry + non-jewelry, FP + discounted, etc.)
   • The achievement threshold (100%) is checked at BILL level, not item level
   • When a bill pushes achievement over 100%, ALL qualifying items in that bill
     receive the 2% bonus commission
   • This ensures items sold together in the same transaction are treated consistently

   JEWELRY IDENTIFICATION RULES:
   An item is classified as jewelry (is_jewelry = 1) if ANY of the following conditions is true:
   • Department is 'WJEW' (Women's Jewelry)
   • Department is 'MJEW' (Men's Jewelry)
   • Vendor code is in the jewelry vendor list:
     - ATS (Atelier Swarovski)
     - VIS (Visionnaire)
     - LUI (Luigi)
     - NAN (Nanan)
     - NAK (Nakard)
     - ROM (Rosa Maria)
     - SPK (Spark)
     - TED (Tedora)
     - BRT (Bartorelli)

   Note: An item is jewelry if it meets ANY of these criteria (OR logic).
   All other items are classified as non-jewelry (is_jewelry = 0).

8. OUTPUT STRUCTURE:

   Returns a pandas DataFrame with the following columns:

   • employee_code: String - HR code (not employee_id which is sid in retailpro)
   • fullname: String - Employee's full name
   • store_code: String - Store identifier (e.g., "RHN", "HBT", "HDG")
   • commission_fp: Number - Full-price commission from personal (non-jewelry) sales in tiers 1-3
   • commission_discount: Number - Discounted commission from personal (non-jewelry) sales in tiers 1-3
   • commission_jewelry: Number - Total jewelry commission (all vendors)
   • commission_vhernier: Number - Commission from VHN vendor jewelry
   • commission_rosa_maria: Number - Commission from ROM EARRINGS (Rosa Maria earrings)
   • commission_100_and_below_fp: Number - FP portion of commission_100_and_below (same as commission_fp)
   • commission_100_and_below_discount: Number - Discount portion of commission_100_and_below (same as commission_discount)
   • commission_100_and_below: Number - Total commission from tiers 1-3 (commission_fp + commission_discount + commission_jewelry)
   • commission_over_100: Number - Commission from tier 4 (over 100% bonus) only
   • total: Number - Total commission (commission_100_and_below + commission_over_100)

   NOTES:
   • commission_fp + commission_discount = personal commission from non-jewelry sales (tiers 1-3)
   • commission_jewelry includes:
     - TVL/TIT flat rate (500,000 VND per item)
     - ROM EARRINGS 3% (commission_rosa_maria)
     - VHN 1% (commission_vhernier)
     - Other vendors 2%
   • commission_100_and_below_fp = commission_fp (FP portion of commission at 100% and below)
   • commission_100_and_below_discount = commission_discount (discount portion of commission at 100% and below)
   • commission_100_and_below = commission_fp + commission_discount + commission_jewelry
   • commission_over_100 = Tier 4 bonus (2% on full-price, non-jewelry, non-TIT/TVL items after reaching 100% target)
   • total = commission_100_and_below + commission_over_100
   • Employees who don't reach 50% threshold will have all commission values as 0
